
some example of using mbedtls lib

openssl example
1. generate a pair new rsa keys
>openssl genpkey -algorithm rsa -out rsa_private.key
2. generate a certificate sign request
>openssl req -new -key rsa_private.key -out server.csr
3. sign self the certificate sign request and generate a cert
>openssl x509 -req -days 365 -in server.csr -signkey rsa_private.key -out mycert.crt

another example
>openssl genrsa -out ca.key 2048
>openssl req -new -key ca.key -out ca.csr
>openssl x509 -req -days 365 -in ca.csr -signkey ca.key -out ca.crt

query serial
>openssl x509 -in ca.crt -noout -serial
query name and owner
>openssl x509 -in ca.crt -noout -subject
query md5 info
>openssl x509 -in ca.crt -noout -fingerprint
query sha info
>openssl x509 -sha1 -in ca.crt -noout -fingerprint

transform between der and pem
>openssl x509 -inform pem -in certificate.pem -outform der -out certificate.der
>openssl x509 -inform der -in certificate.der -outform pem -out certificate.pem

sign a csr
>openssl genrsa -out server.key 1024
>openssl req -new -key server.key -out server.csr
>openssl x509 -req -days 3000 -sha1 -extensions v3_req -CA ca.crt -CAkey ca.key -CAserial ca.srl -CAcreateserial -in server.csr -out server.crt
-CA: 指定CA证书的路径
-CAkey: 指定证书的私钥路径
-CAserial: 指定证书序列号文件的路径
-CAcreateserial: 表示创建证书序列号文件(即上方提到的serial文件)，创建的序列号文件默认名称为-CA，指定的证书名称后加.srl后缀

cert verify, check signed success or not
>openssl verify -CAfile ca.crt server.crt

example 
>openssl genpkey -algorithm rsa -out rsa_private.key
>openssl rsa -pubout -in rsa_private.key -out rsa_pub.key

encrypt and decrypt file
>openssl rsautl -encrypt -in text.txt -inkey rsa_pub.key -pubin -out text.en
>openssl rsautl -decrypt -in text.en -inkey rsa_private.key
or
>openssl pkeyutl -encrypt -in text.txt -inkey rsa_pub.key -pubin -out text.en
>openssl pkeyutl -decrypt -in text.en -inkey rsa_private.key

sign and verify file
>openssl rsautl -sign -in text.txt -inkey rsa_private.key -out text.sign
>openssl rsautl -verify -in text.sign -inkey rsa_pub.key -pubin
or 
>openssl pkeyutl -sign -in text.txt -inkey rsa_private.key -out text.sign
>openssl pkeyutl -verify -in text.txt -sigfile text.sign -inkey rsa_pub.key -pubin
